\begin{center}
\begin{tikzpicture}[
  every node/.style={circle, draw, thick, minimum size=8mm},
  level/.style={sibling distance=40mm/#1},
  level distance=15mm,
  ->, >=stealth
]

  child {node (a) {30}
    child {node {20}}
    child {node {40}}
  }
  child {node (b) {70}
    child {node {60}}
    child[missing]
  };
\draw[red, dashed, <-] (a) -- ++(-2,0.8) node[draw=none, left] {left subtree};
\end{tikzpicture}
