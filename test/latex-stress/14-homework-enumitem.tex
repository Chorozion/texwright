\begin{problem}{1}
Evaluate the following.
\begin{enumerate}[label=(\alph*)]
  \item $\displaystyle\int_0^1 x^2\,dx$
  \item $\displaystyle\oint_C \vec{F}\cdot d\vec{r}$ and $\iint_D f\,dA$ and $\oiint_S \vec{F}\cdot d\vec{S}$
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}[label=(\alph*), noitemsep]
  \item $\frac{1}{3}$, also written $\sfrac{1}{3}$ or $\nicefrac{1}{3}$.
  \item By Green's theorem, $\oint_C = \iint_D (\partial_x Q - \partial_y P)\,dA$.
\end{enumerate}
\end{solution}
