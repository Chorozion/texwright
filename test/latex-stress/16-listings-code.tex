\begin{lstlisting}[language=Python, caption={Fibonacci in Python}, label={lst:fib}]
def fib(n):
    # iterative version
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a
\end{lstlisting}
